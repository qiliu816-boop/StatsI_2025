\section{Q4: Residual-on-Residual (Add-Variable) Regression}
Let $res\_q1$ be residuals from regressing \textit{voteshare} on \textit{difflog}, and $res\_q2$ be residuals from regressing \textit{presvote} on \textit{difflog}. We estimate:
\begin{equation*}
res\_q1 = \delta_0 + \delta_1 \, res\_q2 + \varepsilon.
\end{equation*}
Fitted equation:
\begin{equation*}
\widehat{res\_q1} = -0.000000 + 0.256877 \cdot res\_q2.
\end{equation*}

\section{Q5: Multiple Regression and FWL Check}
We estimate:
\begin{equation*}
\widehat{voteshare} = \theta_0 + \theta_1 \, difflog + \theta_2 \, presvote.
\end{equation*}
Fitted equation:
\begin{equation*}
\widehat{voteshare} = 0.4486 + 0.0355 \cdot difflog + 0.2569 \cdot presvote.
\end{equation*}

\subsection*{FWL Equality (Numerical Verification)}
The coefficient on \textit{presvote} in Q5 must equal the slope in Q4 for $res\_q2$.
\begin{table}[h!]
\centering
\caption{FWL comparison: Q4 residual-on-residual vs Q5 multiple regression (coefficient on \textit{presvote}).}
\label{tab:fwl}
\begin{tabular}{lrrrr}
\toprule
Model & Estimate & Std. Error & t-statistic & p-value \\
\midrule
Q4 residual-on-residual & 0.2569 & 0.0118 & 21.8398 & 0.0000 \\
Q5 multiple OLS & 0.2569 & 0.0118 & 21.8363 & 0.0000 \\
\bottomrule
\end{tabular}
\end{table}

\paragraph{Interpretation.} After controlling for \textit{difflog}, \textit{presvote} remains a strong predictor of \textit{voteshare}. The identical entries across Q4 and Q5 validate the Frisch--Waugh--Lovell theorem.
